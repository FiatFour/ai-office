<script type="text/x-dc" data-dc-script data-props='{"$preview":{"width":1440,"height":980}}'>
class Component extends DCLogic {
  constructor(props) {
    super(props);
    this.state = { lead: 'claude', count: 4, rt: { 0: 'Claude', 1: 'Codex', 2: 'Claude', 3: 'Codex', 4: 'Claude', 5: 'Codex' }, stack: { 'Next.js': true, 'Go': true, 'Kafka': true, 'IBM MQ': true, 'Oracle': true, 'Clerk': true, 'Supabase': true, 'Stripe': true } };
  }
  renderVals() {
    var self = this;
    var s = this.state;
    var leadDefs = [
      { id: 'claude', name: 'CLAUDE CODE', desc: 'Careful planner', hair: '#4a2f27', skin: '#f2c79b', shirt: '#d97757', tag: 'Claude account' },
      { id: 'codex', name: 'CODEX', desc: 'Fast splitter', hair: '#1b1730', skin: '#c68642', shirt: '#42a5f5', tag: 'ChatGPT account' }
    ];
    var leads = leadDefs.map(function (l) {
      var on = l.id === s.lead;
      return Object.assign({}, l, { pressed: on ? 'true' : 'false', ring: on ? '#ffca28' : '#454b55', glow: on ? '0 0 0 4px #1f2328, 0 0 0 8px #ffca28' : 'none', anim: on ? 'bob' : '', tagBg: on ? '#ffca28' : '#9aa1ab', pick: function () { self.setState({ lead: l.id }); } });
    });
    var names = ['Byte', 'Pixel', 'Nova', 'Echo', 'Bit', 'Glitch'];
    var looks = [['#4a2f27', '#e0a878', '#2fbf71'], ['#e2557a', '#f2c79b', '#ff9f43'], ['#1b1730', '#c68642', '#42a5f5'], ['#9b9b9b', '#f2c79b', '#b06cff'], ['#ffca28', '#e0a878', '#ef5350'], ['#2fbf71', '#f2c79b', '#6b7280']];
    var slots = names.map(function (n, i) {
      var filled = i < s.count;
      var rt = s.rt[i];
      return { name: n, filled: filled, empty: !filled, hair: looks[i][0], skin: looks[i][1], shirt: looks[i][2], border: filled ? '#6b7280' : '#454b55', borderStyle: filled ? 'solid' : 'dashed', runtime: rt, rtBg: rt === 'Claude' ? '#ff9f43' : '#42a5f5', flip: function () { var r = Object.assign({}, self.state.rt); r[i] = rt === 'Claude' ? 'Codex' : 'Claude'; self.setState({ rt: r }); } };
    });
    var stackDefs = [['Next.js', '#f5f3ec'], ['Go', '#00add8'], ['Kafka', '#ffca28'], ['IBM MQ', '#4589ff'], ['Oracle', '#c74634'], ['Clerk', '#7c5cff'], ['Supabase', '#2fbf71'], ['Stripe', '#42a5f5'], ['Tailwind', '#38bdf8'], ['Prisma', '#9aa1ab']];
    var stack = stackDefs.map(function (d) {
      var on = !!s.stack[d[0]];
      return { name: d[0], color: d[1], pressed: on ? 'true' : 'false', border: on ? '#ffca28' : '#454b55', fg: on ? '#f5f3ec' : '#c9cdd4', state: on ? 'Equipped' : 'Not equipped', toggle: function () { var x = Object.assign({}, self.state.stack); x[d[0]] = !on; self.setState({ stack: x }); } };
    });
    var eq = stack.filter(function (x) { return x.pressed === 'true'; }).length;
    var leadName = s.lead === 'claude' ? 'Claude Code' : 'Codex';
    return {
      leads: leads,
      slots: slots,
      count: s.count,
      inc: function () { self.setState({ count: Math.min(6, self.state.count + 1) }); },
      dec: function () { self.setState({ count: Math.max(1, self.state.count - 1) }); },
      stack: stack,
      checks: [
        { label: 'Git repo with a GitHub or GitLab remote', mark: 'OK', color: '#66bb6a', bad: false },
        { label: 'gh (GitHub) or glab (GitLab) signed in', mark: 'OK', color: '#66bb6a', bad: false },
        { label: 'Docker Sandboxes ready', mark: 'OK', color: '#66bb6a', bad: false },
        { label: 'Network policy set', mark: '!!', color: '#ef5350', bad: true, fix: 'sbx policy init balanced' },
        { label: 'tmux installed', mark: 'OK', color: '#66bb6a', bad: false }
      ],
      summary: 'Lead: ' + leadName + ' · ' + s.count + ' workers · ' + eq + ' tools equipped'
    };
  }
}
